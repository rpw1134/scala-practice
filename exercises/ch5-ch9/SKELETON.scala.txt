  private var passed = 0
  private var failed = 0

  private def check[A](label: String, actual: => A, expected: A): Unit = {
    val outcome =
      try if (actual == expected) Right(()) else Left(actual.toString)
      catch { case e: Throwable => Left(s"threw ${e.getClass.getSimpleName}: ${e.getMessage}") }

    outcome match {
      case Right(_) => passed += 1; println(s"  ok   $label")
      case Left(got) =>
        failed += 1
        println(s"  FAIL $label")
        println(s"         expected: $expected")
        println(s"         got:      $got")
    }
  }

  private def timed[A](f: => Future[A]): (A, Long) = {
    val t0 = System.currentTimeMillis()
    val r = Await.result(f, 10.seconds)
    (r, System.currentTimeMillis() - t0)
  }

    // Task 7

    check("describe ok",    describeFailure(safeParse("42")), "parsed 42")
    check("describe bad",   describeFailure(safeParse("4x")), "failed: bad input: 4x")

    // Task 8 — a fixed clock, passed explicitly, overrides the implicit one

    check("render receipt", render(receiptFor(successes.head)),
                            "receipt: 2 x widget = $50.00")

    // Task 10
    val (seqResult, seqMs) = timed(checkoutSequential(successes.head))
    check("sequential value", seqResult, "res-widget-2|chg-alice-5000")
    check("sequential is ~600ms", seqMs >= 550, true)

    val (parResult, parMs) = timed(checkoutParallel(successes.head))
    check("parallel value", parResult, "res-widget-2|8")
    check("parallel is ~300ms", parMs < 500, true)

    val (allResult, allMs) = timed(reserveAll(successes))
    check("reserveAll value", allResult, List("res-widget-2", "res-gizmo-2"))
    check("reserveAll is concurrent", allMs < 500, true)

    // Task 11
    check("firstSuccess",       firstSuccess(validated).map(_.sku), Some("widget"))
    check("firstSuccess empty", firstSuccess(failures.map(Left(_))), None)
